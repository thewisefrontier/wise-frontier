function clusterArticles(articles) { const clusters = []; const used = new Set(); articles.forEach((a, i) => { if (used.has(i)) return; const wordsA = getKeywords(a.title_en || ''); const group = [a]; used.add(i); articles.forEach((b, j) => { if (i === j || used.has(j)) return; const wordsB = getKeywords(b.title_en || ''); const overlap = wordsA.filter(w => wordsB.includes(w)).length; if (overlap >= 2) { group.push(b); used.add(j); } }); // 클러스터 내부: 최신순 group.sort((x, y) => (y.created_at || '') > (x.created_at || '') ? 1 : -1); clusters.push(group); }); // 클러스터끼리: 각 클러스터 대표(첫 번째) 기사 최신순 clusters.sort((a, b) => (b[0].created_at || '') > (a[0].created_at || '') ? 1 : -1); return clusters; }
